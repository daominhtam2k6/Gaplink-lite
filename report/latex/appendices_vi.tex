\section{Vòng lặp Huấn luyện}
Mỗi bước lấy một batch sự kiện che khuất tổng hợp với tỉ lệ sự kiện khó tăng dần, và tối ưu mất mát gán cùng mất mát Gaussian bằng AdamW, khởi động tuyến tính và giảm theo cosine.

\begin{lstlisting}[language=Python, caption={Một bước huấn luyện (rút gọn)}, label={lst:train}]
for step in range(1, total_steps + 1):
    hard_prob = hp_start + (hp_end - hp_start) * min(1.0, step / total_steps / 0.6)
    lr = base_lr * step / warmup_steps if step <= warmup_steps else \
         base_lr * 0.5 * (1 + math.cos(math.pi * (step - warmup_steps) / (total_steps - warmup_steps)))
    for pg in opt.param_groups:
        pg["lr"] = lr
    batch = pool_ds.sample_batch(batch_size, hard_prob=hard_prob, rng=rng)
    losses = model.compute_loss(batch, w_assign=1.0, w_nll=0.5)   # L_assign + 0.5 L_NLL
    opt.zero_grad(set_to_none=True)
    losses["loss"].backward()
    torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
    opt.step()
\end{lstlisting}

\section{Sinkhorn Miền Log với Dustbin}
\begin{lstlisting}[language=Python, caption={Vận chuyển tối ưu Sinkhorn miền log}, label={lst:sinkhorn}]
def log_sinkhorn(S, z_t, z_c, trk_mask, cand_mask, iters=20):
    B, N, M = S.shape
    neg_inf = -1e9
    N_act = trk_mask.float().sum(dim=1).clamp(min=1.0)
    M_act = cand_mask.float().sum(dim=1).clamp(min=1.0)
    valid_pair = trk_mask[:, :, None] & cand_mask[:, None, :]
    S_m = torch.where(valid_pair, S, S.new_full(S.shape, neg_inf))
    zt_m = torch.where(trk_mask, z_t, z_t.new_full(z_t.shape, neg_inf))
    zc_m = torch.where(cand_mask, z_c, z_c.new_full(z_c.shape, neg_inf))
    top = torch.cat([S_m, zt_m.unsqueeze(-1)], dim=2)
    bot = torch.cat([zc_m, S.new_zeros((B, 1))], dim=1).unsqueeze(1)
    log_alpha = torch.cat([top, bot], dim=1)
    log_r = S.new_zeros((B, N + 1))
    log_r[:, :N] = torch.where(trk_mask, 0.0, neg_inf)
    log_r[:, N] = torch.log(M_act)
    log_c = S.new_zeros((B, M + 1))
    log_c[:, :M] = torch.where(cand_mask, 0.0, neg_inf)
    log_c[:, M] = torch.log(N_act)
    for _ in range(iters):
        log_alpha = log_alpha - torch.logsumexp(log_alpha, dim=2, keepdim=True) + log_r.unsqueeze(2)
        log_alpha = log_alpha - torch.logsumexp(log_alpha, dim=1, keepdim=True) + log_c.unsqueeze(1)
    return log_alpha
\end{lstlisting}

\section{Quyết định Tái liên kết}
\begin{lstlisting}[language=Python, caption={Chỉ chấp nhận liên kết khi vượt ngưỡng và vượt dustbin}, label={lst:decide}]
def decide(self, ev):
    N, M = len(ev["t_lost"]), len(ev["cand"])
    out = np.full((N,), -1, dtype=np.int32)          # -1 = keep waiting (dustbin)
    if N == 0 or M == 0:
        return out
    P = self.infer_event(ev)["P"]
    P_real, P_dust = P[:N, :M], P[:N, M]
    rows, cols = linear_sum_assignment(-P_real)
    for r, c in zip(rows, cols):
        if P_real[r, c] >= self.tau and P_real[r, c] > P_dust[r]:
            out[r] = int(c)
    return out
\end{lstlisting}

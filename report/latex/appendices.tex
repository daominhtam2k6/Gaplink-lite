\chapter{Source Code and Implementation Details}
\label{app:code}

This appendix provides the core implementation of the log-domain Sinkhorn optimal transport layer with track and candidate dustbins (\texttt{src/model/sinkhorn.py}) and the GapLink-Lite forward and loss computation (\texttt{src/model/gaplink.py}).

\begin{lstlisting}[language=Python, caption={Log-Domain Sinkhorn Optimal Transport with Dustbin Marginals (src/model/sinkhorn.py)}, label={lst:sinkhorn}]
import torch

def log_sinkhorn(
    S: torch.Tensor,
    z_t: torch.Tensor,
    z_c: torch.Tensor,
    trk_mask: torch.Tensor,
    cand_mask: torch.Tensor,
    iters: int = 20,
) -> torch.Tensor:
    B, N, M = S.shape
    neg_inf = -1e9
    tm_f = trk_mask.float()
    cm_f = cand_mask.float()
    N_act = tm_f.sum(dim=1).clamp(min=1.0)
    M_act = cm_f.sum(dim=1).clamp(min=1.0)

    valid_pair = trk_mask[:, :, None] & cand_mask[:, None, :]
    S_masked = torch.where(valid_pair, S, S.new_full(S.shape, neg_inf))
    zt_masked = torch.where(trk_mask, z_t, z_t.new_full(z_t.shape, neg_inf))
    zc_masked = torch.where(cand_mask, z_c, z_c.new_full(z_c.shape, neg_inf))

    top = torch.cat([S_masked, zt_masked.unsqueeze(-1)], dim=2)
    bot_c = torch.cat([zc_masked, S.new_zeros((B, 1))], dim=1).unsqueeze(1)
    log_alpha = torch.cat([top, bot_c], dim=1)

    log_r = S.new_zeros((B, N + 1))
    log_r[:, :N] = torch.where(trk_mask, 0.0, neg_inf)
    log_r[:, N] = torch.log(M_act)

    log_c = S.new_zeros((B, M + 1))
    log_c[:, :M] = torch.where(cand_mask, 0.0, neg_inf)
    log_c[:, M] = torch.log(N_act)

    for _ in range(iters):
        log_alpha = log_alpha - torch.logsumexp(log_alpha, dim=2, keepdim=True) + log_r.unsqueeze(2)
        log_alpha = log_alpha - torch.logsumexp(log_alpha, dim=1, keepdim=True) + log_c.unsqueeze(1)
    return log_alpha
\end{lstlisting}

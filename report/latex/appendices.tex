\begin{lstlisting}[language=Python, caption={Log-Domain Sinkhorn Optimal Transport (src/model/sinkhorn.py)}, label={lst:sinkhorn}]
def log_sinkhorn(S, z_t, z_c, trk_mask, cand_mask, iters=20):
    B, N, M = S.shape
    neg_inf = -1e9
    N_act = trk_mask.float().sum(dim=1).clamp(min=1.0)
    M_act = cand_mask.float().sum(dim=1).clamp(min=1.0)
    valid_pair = trk_mask[:, :, None] & cand_mask[:, None, :]
    S_m = torch.where(valid_pair, S, S.new_full(S.shape, neg_inf))
    zt_m = torch.where(trk_mask, z_t, z_t.new_full(z_t.shape, neg_inf))
    zc_m = torch.where(cand_mask, z_c, z_c.new_full(z_c.shape, neg_inf))
    top = torch.cat([S_m, zt_m.unsqueeze(-1)], dim=2)
    bot = torch.cat([zc_m, S.new_zeros((B, 1))], dim=1).unsqueeze(1)
    log_alpha = torch.cat([top, bot], dim=1)
    log_r = S.new_zeros((B, N + 1))
    log_r[:, :N] = torch.where(trk_mask, 0.0, neg_inf)
    log_r[:, N] = torch.log(M_act)
    log_c = S.new_zeros((B, M + 1))
    log_c[:, :M] = torch.where(cand_mask, 0.0, neg_inf)
    log_c[:, M] = torch.log(N_act)
    for _ in range(iters):
        log_alpha = log_alpha - torch.logsumexp(log_alpha, dim=2, keepdim=True) + log_r.unsqueeze(2)
        log_alpha = log_alpha - torch.logsumexp(log_alpha, dim=1, keepdim=True) + log_c.unsqueeze(1)
    return log_alpha
\end{lstlisting}

\begin{lstlisting}[language=Python, caption={GapLinkLite Forward \& Joint Loss (src/model/gaplink.py)}, label={lst:gaplink}]
class GapLinkLite(nn.Module):
    def forward(self, batch):
        h_i, g_emb, last_box = self.motion_enc(batch["hist"], batch["hist_len"], batch["gap"], batch["W"], batch["H"], gap_aware=self.gap_aware)
        ctx_i = self.ctx_enc(batch["ctx_lost"], batch["ctx_conf"], batch["ctx_mask_lost"], batch["ctx_mask_conf"], use_ctx=self.use_ctx)
        mu, logvar = self.predictor(h_i, ctx_i, batch["hist"], batch["hist_len"], batch["gap"], gap_aware=self.gap_aware)
        S, z_t, z_c = self.scorer(h_i=h_i, ctx_i=ctx_i, g_emb=g_emb, last_box=last_box, mu=mu, logvar=logvar,
                                  track_color=batch["track_color"], cand=batch["cand"], cand_hist=batch["cand_hist"],
                                  side=batch.get("side", torch.zeros_like(batch["gap"])), W=batch["W"], H=batch["H"],
                                  use_color=self.use_color, training=self.training)
        logP = log_sinkhorn(S, z_t, z_c, batch["trk_mask"], batch["cand_mask"], iters=self.sinkhorn_iters)
        pred_box, sigma_px = decode_predicted_boxes(last_box, mu, logvar)
        return {"logP": logP, "P": torch.exp(logP), "mu": mu, "logvar": logvar, "pred_box": pred_box, "sigma_px": sigma_px}
\end{lstlisting}
